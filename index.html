async function executarIA() { const linkVideo = document.getElementById('input-link-video').value.trim(); const videoFiles = document.getElementById('input-file-video').files; const overlay = document.getElementById('processing-overlay'); const titleEl = document.getElementById('processing-title'); // VALIDAÇÃO LOCAL: Impede o sistema de tentar rodar se nada foi selecionado if (videoFiles.length === 0 && linkVideo === '') { falar("Por favor, selecione um arquivo ou cole um link antes de continuar."); alert("Selecione um arquivo ou cole um link válido!"); return; } falar("Iniciando varredura do lote de vídeos. Aguarde."); overlay.style.display = 'flex'; setTimeout(() => overlay.classList.remove('opacity-0'), 10); progressoAtual = 0; progressoAlvo = 0; clearTimeout(loopProgresso); animarProgresso(); iniciarAnimacoesOverlay(); try { let formData = new FormData(); // Formata os arquivos corretamente para envio em lote if (videoFiles.length > 0) { for (let i = 0; i < videoFiles.length; i++) { formData.append("video_files", videoFiles[i]); } } // Só envia o link se ele existir if (linkVideo !== '') { formData.append("link_video", linkVideo); } await fetch(`${API_BASE}/processar-video`, { method: 'POST', headers: { 'ngrok-skip-browser-warning': 'true' }, body: formData }); let poll = setInterval(async () => { try { const res = await fetch(`${API_BASE}/status`, { headers: { 'ngrok-skip-browser-warning': 'true' } }); const statusData = await res.json(); progressoAlvo = statusData.progresso; if (titleEl && statusData.mensagem) { titleEl.innerHTML = ` ${statusData.mensagem}`; } if (statusData.erro) { clearInterval(poll); pararAnimacoesOverlay(); overlay.classList.add('opacity-0'); setTimeout(() => overlay.style.display = 'none', 500); falar("Erro crítico no processamento."); alert(statusData.erro); } if(statusData.progresso === 100) { clearInterval(poll); progressoAlvo = 100; setTimeout(() => { pararAnimacoesOverlay(); tocarSomSucesso(); overlay.classList.add('opacity-0'); setTimeout(() => overlay.style.display = 'none', 500); const list = document.getElementById('download-buttons-list'); list.innerHTML = ''; statusData.modulos.forEach(item => { window.arquivosGerados[item.arquivo] = item; list.innerHTML += ` `; }); document.getElementById('download-zone').classList.remove('hidden'); falar("Processamento finalizado com sucesso. Lote disponível para download."); }, 1200); } } catch (e) { console.error("Erro no polling:", e); } }, 2000); } catch (error) { pararAnimacoesOverlay(); overlay.classList.add('opacity-0'); setTimeout(() => overlay.style.display = 'none', 300); falar("Falha na conexão com o sistema."); } }
